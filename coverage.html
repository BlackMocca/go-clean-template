type psqlRepository struct {
        client *psql.Client
}

func NewPsqlRepository(client *psql.Client) Repository {
        return &psqlRepository{client: client}
}

// FindByID retrieves a user by their unique ID. Returns an error if not found.
func (p psqlRepository) FindByID(ctx context.Context, idOrEmail string) (*models.User, error) {
        uid, _ := uuid.Parse(idOrEmail)
        sql := `
                SELECT 
                        users.*,
                        COALESCE(
                                json_agg(row_to_json(permissions)) FILTER (WHERE permissions.id IS NOT NULL),
                                '[]'::json
                        ) AS permissions
                FROM 
                        users
                LEFT JOIN
                        user_permissions
                ON
                        users.id = user_permissions.user_id
                LEFT JOIN
                        permissions
                ON
                        permissions.id = user_permissions.permission_id
                WHERE 
                        users.deleted_at IS NULL
                        AND
                        (users.id = ? or users.email = ?)
                GROUP BY users.id
        ORDER BY users.id ASC
        `
        sql = sqlx.Rebind(sqlx.DOLLAR, sql)
        stmt, err := p.client.GetClient().PreparexContext(ctx, sql)
        if err != nil {
                return nil, err
        }
        defer stmt.Close()

        rows, err := stmt.Unsafe().QueryxContext(ctx, uid, idOrEmail)
        if err != nil {
                return nil, err
        }
        defer rows.Close()

        users, err := p.orm(ctx, rows, nil)
        if err != nil {
                return nil, err
        }

        if len(users) > 0 {
                return users[0], nil
        }

        return nil, nil
}

func (p psqlRepository) upsertUser(ctx context.Context, tx *sqlx.Tx, user *models.User) error {
        sql := `
                INSERT INTO users (id, name, email, role, created_at, updated_at, deleted_at)
                VALUES (?, ?, ?, ?::user_role, ?,?, NULL)
                ON CONFLICT (id)
                DO UPDATE SET
                        email = EXCLUDED.email,
                        name = EXCLUDED.name,
                        updated_at = EXCLUDED.updated_at;
        `

        sql = sqlx.Rebind(sqlx.DOLLAR, sql)
        stmt, err := tx.PreparexContext(ctx, sql)
        if err != nil {
                return err
        }
        defer stmt.Close()

        _, merr := stmt.ExecContext(ctx,
                user.ID,
                user.Name,
                user.Email,
                user.Role,
                user.CreatedAt,
                user.UpdatedAt,
        )
        return merr
}

func (p psqlRepository) deleteUserPermission(ctx context.Context, tx *sqlx.Tx, userId uuid.UUID) error {
        sql := `
                DELETE FROM user_permissions
                WHERE
                        user_id = ?
        `

        sql = sqlx.Rebind(sqlx.DOLLAR, sql)
        stmt, err := tx.PreparexContext(ctx, sql)
        if err != nil {
                return err
        }
        defer stmt.Close()

        _, merr := stmt.ExecContext(ctx,
                userId,
        )
        return merr
}

func (p psqlRepository) insertPermissions(ctx context.Context, tx *sqlx.Tx, userId uuid.UUID, permissions []uuid.UUID) error {
        sql := `
                INSERT INTO user_permissions (user_id, permission_id)
                VALUES (?, ?)
        `

        sql = sqlx.Rebind(sqlx.DOLLAR, sql)
        stmt, err := tx.PreparexContext(ctx, sql)
        if err != nil {
                return err
        }
        defer stmt.Close()

        for _, permissionId := range permissions {
                if _, err := stmt.ExecContext(ctx, userId, permissionId); err != nil {
                        return err
                }
        }
        return nil
}

// Save persists a user to storage. If the user has no ID set (ID == 0),
// the implementation auto-generates one.
func (p psqlRepository) Save(ctx context.Context, user *models.User) error {
        var tx, err = p.client.GetClient().Beginx()
        if err != nil {
                return err
        }
        defer tx.Rollback()

        if err := p.upsertUser(ctx, tx, user); err != nil {
                return err
        }
        if err := p.deleteUserPermission(ctx, tx, user.ID); err != nil {
                return err
        }
        if err := p.insertPermissions(ctx, tx, user.ID, models.Permissions(user.Permissions.Data()).GetIds()); err != nil {
                return err
        }

        return tx.Commit()
}

// Delete removes a user by their ID. Returns an error if the user does not exist.
func (p psqlRepository) Delete(ctx context.Context, id uuid.UUID) error {
        var tx, err = p.client.GetClient().Beginx()
        if err != nil {
                return err
        }
        defer tx.Rollback()

        sql := `
        UPDATE users
        SET deleted_at = NOW()
        WHERE id = ?
        `

        sql = sqlx.Rebind(sqlx.DOLLAR, sql)
        stmt, err := tx.PreparexContext(ctx, sql)
        if err != nil {
                return err
        }
        defer stmt.Close()

        if _, merr := stmt.ExecContext(ctx, id); merr != nil {
                return merr
        }

        return tx.Commit()
}

func (p psqlRepository) whereConds(params ListParams) ([]string, []any) {
        var conds = []string{}
        var vals = []any{}

        if v := params.Searchword; v != "" {
                sql := `users.email LIKE CONCAT('%', ?::text, '%')`
                conds = append(conds, sql)
                vals = append(vals, v)
        }

        if v := params.Role; v.Valid() {
                sql := `permissions.name = ?`
                conds = append(conds, sql)
                vals = append(vals, string(params.Role))
        }

        return conds, vals
}

// List returns all users currently stored. Never returns nil, may return an empty slice.
func (p psqlRepository) List(ctx context.Context, params ListParams) (models.Paginator[[]*models.User], error) {
        var paginator = models.Paginator[[]*models.User]{Page: params.Page, PageSize: params.PageSize}
        var where string
        var conds, vals = p.whereConds(params)
        if len(conds) > 0 {
                where = " AND " + strings.Join(conds, " AND ")
        }

        templateCountSQL := fmt.Sprintf(`
                SELECT 
                        COUNT(DISTINCT users.id) as total_documents
                FROM 
                        users
                LEFT JOIN
                        user_permissions
                ON
                        users.id = user_permissions.user_id
                LEFT JOIN
                        permissions
                ON
                        permissions.id = user_permissions.permission_id
                WHERE 
                        users.deleted_at IS NULL
                        %s
        `,
                where,
        )

        templateDataSQL := fmt.Sprintf(`
                SELECT 
                        users.*,
                        COALESCE(
                                json_agg(row_to_json(permissions)) FILTER (WHERE permissions.id IS NOT NULL),
                                '[]'::json
                        ) AS permissions
                FROM 
                        users
                LEFT JOIN
                        user_permissions
                ON
                        users.id = user_permissions.user_id
                LEFT JOIN
                        permissions
                ON
                        permissions.id = user_permissions.permission_id
                WHERE 
                        users.deleted_at IS NULL
                        %s
                GROUP BY users.id
                LIMIT %d
                OFFSET %d
        `,
                where,
                paginator.Limit(),
                paginator.Offset(),
        )

        var execute = func(ctx context.Context, query string) (models.Paginator[[]*models.User], error) {
                query = sqlx.Rebind(sqlx.DOLLAR, query)
                stmt, err := p.client.GetClient().PreparexContext(ctx, query)
                if err != nil {
                        return paginator, err
                }
                defer stmt.Close()

                rows, err := stmt.Unsafe().QueryxContext(ctx, vals...)
                if err != nil {
                        return paginator, err
                }
                defer rows.Close()

                users, err := p.orm(ctx, rows, &paginator)
                if err != nil {
                        return paginator, nil
                }

                paginator.SetDocument(users)
                return paginator, nil
        }

        pagiantor, err := execute(ctx, templateDataSQL)
        if err != nil {
                return pagiantor, err
        }
        totalDocuments, err := execute(ctx, templateCountSQL)
        if err != nil {
                return pagiantor, err
        }

        pagiantor.SetTotalDocument(totalDocuments.TotalDocument)
        return pagiantor, nil
}

func (p psqlRepository) orm(ctx context.Context, rows *sqlx.Rows, paginator *models.Paginator[[]*models.User]) ([]*models.User, error) {
        var ptrs = make([]*models.User, 0)
        type tmp struct {
                TotalDocument zero.Int64 `db:"total_documents"`
                models.User
        }

        for rows.Next() {
                var ptr = tmp{}
                if err := rows.StructScan(&ptr); err != nil {
                        return nil, err
                }

                if paginator != nil {
                        paginator.SetTotalDocument(ptr.TotalDocument.ValueOrZero())
                }

                ptrs = append(ptrs, &ptr.User)
        }

        if paginator != nil {
                paginator.SetDocument(paginator.Documents)
        }

        return ptrs, nil
}

// Package user defines the use-case layer for user operations.
// It depends on a Repository interface so that real and mock
// implementations can be swapped during testing.
package user

import (
        "context"
        "errors"

        "example/constants"
        "example/models"

        "github.com/google/uuid"
)

// userUsecase implements the Usecase interface using a Repository for data access.
type userUsecase struct {
        repo Repository // injected dependency; nil-safe validation is done in each method
}

// NewUserUsecase constructs a new Usecase with the given repository.
// Pass a mock Repository in tests to isolate business logic from data access.
func NewUserUsecase(repo Repository) Usecase {
        return &userUsecase{repo: repo}
}

// GetUserByID validates the ID and retrieves a single user via the repository.
func (s *userUsecase) GetUserByID(ctx context.Context, id uuid.UUID) (*models.User, error) {
        return s.repo.FindByID(ctx, id.String())
}

// ListUsers delegates to the repository and returns all users.
func (s *userUsecase) ListUsers(ctx context.Context, params ListParams) (models.Paginator[[]*models.User], error) {
        return s.repo.List(ctx, params)
}

func (s *userUsecase) Save(ctx context.Context, user *models.User) error {
        existsUser, err := s.repo.FindByID(ctx, user.Email.ValueOrZero())
        if err != nil {
                return err
        }
        if existsUser != nil {
                return errors.New(string(constants.ApplicationCodeConflictUserEmailWasDuplicate))
        }
        return s.repo.Save(ctx, user)
}

// Delete removes a user by their ID. Returns an error if the user does not exist.
func (s *userUsecase) Delete(ctx context.Context, id uuid.UUID) error {
        user, err := s.repo.FindByID(ctx, id.String())
        if err != nil {
                return err
        }
        if user == nil {
                return errors.New(string(constants.ApplicationCodeConflictUserAlreadyRemoved))
        }
        return s.repo.Delete(ctx, id)
}

        if err := fn.CopyJSON(paginator, &resp.Data); err != nil {

        }

}

// สร้างผู้ใช้ใหม่
// (POST /api/v1/users)
func (h handler) CreateUser(ctx echo.Context) error {
        var rawctx = ctx.Request().Context()
        var reqBody = UpdateUserByIDJSONRequestBody{}
        if err := ctx.Bind(&reqBody); err != nil {
                return middleware.NewResponseError(http.StatusInternalServerError, constants.ApplicationCodeBadRequestInvalidArgument, constants.ApplicationErrorInvalidArgument, err.Error(), nil)
        }

        var user = new(models.User)
        user.ID = uuid.New()
        if err := fn.CopyJSON(reqBody, user); err != nil {
                return middleware.NewResponseError(http.StatusInternalServerError, constants.ApplicationInternalServerError, constants.ApplicationErrorInvalidArgument, err.Error(), nil)
        }
        var permissions = make([]*models.Permission, 0, len(reqBody.PermissionIds))
        for _, permisId := range reqBody.PermissionIds {
                permissions = append(permissions, &models.Permission{Id: uuid.MustParse(permisId.String())})
        }
        user.Permissions.Set(permissions)
        user.CreatedAt = utilModels.NewTimestampFromNow().ToPointer()
        user.UpdatedAt = utilModels.NewTimestampFromNow().ToPointer()

        if err := h.userUs.Save(rawctx, user); err != nil {
                if strings.EqualFold(err.Error(), string(constants.ApplicationCodeConflictUserEmailWasDuplicate)) {
                        return middleware.NewResponseError(http.StatusConflict, constants.ApplicationCodeConflictUserEmailWasDuplicate, constants.ApplicationErrorConflict, fmt.Sprintf("email %s was duplicated", user.Email.ValueOrZero()), nil)
                }
                return middleware.NewResponseError(http.StatusInternalServerError, constants.ApplicationInternalServerError, constants.ApplicationErrorInternalServerError, err.Error(), nil)
        }

        resp := GetUserByIDSuccess{}
        fn.CopyJSON(user, &resp.Data)
        return middleware.NewResponseJSON(resp.Data)
}

// ลบผู้ใช้
// (DELETE /api/v1/users/{id})
func (h handler) DeleteUserByID(ctx echo.Context, id DocumentID) error {
        var rawctx = ctx.Request().Context()

        if err := h.userUs.Delete(rawctx, id); err != nil {
                if strings.EqualFold(err.Error(), string(constants.ApplicationCodeConflictUserAlreadyRemoved)) {
                        return middleware.NewResponseError(http.StatusConflict, constants.ApplicationCodeConflictUserAlreadyRemoved, constants.ApplicationErrorConflict, fmt.Sprintf("user id already removed: %s", id), nil)
                }
                return middleware.NewResponseError(http.StatusInternalServerError, constants.ApplicationInternalServerError, constants.ApplicationErrorInternalServerError, err.Error(), nil)
        }

        return middleware.NewResponseJSON(nil)
}

// This is a simple interface which specifies echo.Route addition functions which
// are present on both echo.Echo and echo.Group, since we want to allow using
// either of them for path registration
type EchoRouter interface {
        CONNECT(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        DELETE(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        GET(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        HEAD(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        OPTIONS(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        PATCH(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        POST(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        PUT(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
        TRACE(path string, h echo.HandlerFunc, m ...echo.MiddlewareFunc) *echo.Route
}

// RegisterHandlers adds each server route to the EchoRouter.
func RegisterHandlers(router EchoRouter, si ServerInterface) {
        RegisterHandlersWithBaseURL(router, si, "")
}

// Registers handlers, and prepends BaseURL to the paths, so that the paths
// can be served under a prefix.
func RegisterHandlersWithBaseURL(router EchoRouter, si ServerInterface, baseURL string) {

        wrapper := ServerInterfaceWrapper{
                Handler: si,
        }

        router.GET(baseURL+"/api/health", wrapper.HealthCheck)
        router.GET(baseURL+"/api/v1/users", wrapper.ListUsers)
        router.POST(baseURL+"/api/v1/users", wrapper.CreateUser)
        router.DELETE(baseURL+"/api/v1/users/:id", wrapper.DeleteUserByID)
        router.GET(baseURL+"/api/v1/users/:id", wrapper.GetUserByID)
        router.PUT(baseURL+"/api/v1/users/:id", wrapper.UpdateUserByID)

}

// GetSwagger returns the content of the embedded swagger specification file
// or error if failed to decode
func decodeSpec() ([]byte, error) {
        zipped, err := base64.StdEncoding.DecodeString(strings.Join(swaggerSpec, ""))
        if err != nil {
                return nil, fmt.Errorf("error base64 decoding spec: %w", err)
        }
        zr, err := gzip.NewReader(bytes.NewReader(zipped))
        if err != nil {
                return nil, fmt.Errorf("error decompressing spec: %w", err)
        }
        var buf bytes.Buffer
        _, err = buf.ReadFrom(zr)
        if err != nil {
                return nil, fmt.Errorf("error decompressing spec: %w", err)
        }

        return buf.Bytes(), nil
}

var rawSpec = decodeSpecCached()

// a naive cached of a decoded swagger spec
func decodeSpecCached() func() ([]byte, error) {
        data, err := decodeSpec()
        return func() ([]byte, error) {
                return data, err
        }
}

// Constructs a synthetic filesystem for resolving external references when loading openapi specifications.
func PathToRawSpec(pathToFile string) map[string]func() ([]byte, error) {
        res := make(map[string]func() ([]byte, error))
        if len(pathToFile) > 0 {
                res[pathToFile] = rawSpec
        }

        return res
}

// GetSwagger returns the Swagger specification corresponding to the generated code
// in this file. The external references of Swagger specification are resolved.
// The logic of resolving external references is tightly connected to "import-mapping" feature.
// Externally referenced files must be embedded in the corresponding golang packages.
// Urls can be supported but this task was out of the scope.
func GetSwagger() (swagger *openapi3.T, err error) {
        resolvePath := PathToRawSpec("")

        loader := openapi3.NewLoader()
        loader.IsExternalRefsAllowed = true
        loader.ReadFromURIFunc = func(loader *openapi3.Loader, url *url.URL) ([]byte, error) {
                pathToFile := url.String()
                pathToFile = path.Clean(pathToFile)
                getSpec, ok := resolvePath[pathToFile]
                if !ok {
                        err1 := fmt.Errorf("path not found: %s", pathToFile)
                        return nil, err1
                }
                return getSpec()
        }
        var specData []byte
        specData, err = rawSpec()
        if err != nil {
                return
        }
        swagger, err = loader.LoadFromData(specData)
        if err != nil {
                return
        }
        return
}
